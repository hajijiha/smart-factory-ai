cmake_minimum_required(VERSION 3.16)
project(factory_world)
find_package(gazebo REQUIRED)
find_package(ament_cmake REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_library(factory_world SHARED world.cpp)
target_include_directories(factory_world PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(factory_world ${GAZEBO_LIBRARIES})
ament_target_dependencies(factory_world gazebo_ros rclcpp std_msgs sensor_msgs)
